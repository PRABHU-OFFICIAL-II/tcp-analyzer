---
  Phase 1 — Check the current state of the VM

  1.1 — Check OS version
  cat /etc/os-release

  1.2 — Check Python availability
  python3 --version; python3.11 --version 2>/dev/null || echo "python3.11 not found"

  1.3 — Check Node.js
  node --version 2>/dev/null || echo "node not found"

  1.4 — Check disk space
  df -h

  1.5 — Check SELinux status
  getenforce

  ---
  Phase 2 — Install system dependencies

  2.1 — Update the system
  sudo dnf update -y

  2.2 — Install Python 3.11
  sudo dnf install -y python3.11 python3.11-devel python3-pip

  2.3 — Install libpcap and tcpdump
  sudo dnf install -y libpcap-devel tcpdump

  2.4 — Add NodeSource RPM repo
  curl -fsSL https://rpm.nodesource.com/setup_20.x | sudo bash -

  2.5 — Install Node.js
  sudo dnf install -y nodejs

  2.6 — Install nginx
  sudo dnf install -y nginx

  2.7 — Install utilities
  sudo dnf install -y git rsync zip unzip

  2.8 — Verify all installs
  python3.11 --version && node --version && npm --version && nginx -v

  ---
  Phase 3 — Transfer the project

  3.1 — Create the app directory
  sudo mkdir -p /opt/tcp-analyzer
  sudo chown $USER:$USER /opt/tcp-analyzer

  3.2 — Transfer from Windows (run this on your Windows machine in WSL or Git Bash)
  rsync -av --progress \
    --exclude='.venv' \
    --exclude='node_modules' \
    --exclude='__pycache__' \
    --exclude='*.pyc' \
    /mnt/c/Users/ppenthoi/Documents/DEV/tcp-analyzer/ \
    root@YOUR_VM_IP:/opt/tcp-analyzer/

  3.3 — Confirm files arrived on VM
  ls -la /opt/tcp-analyzer/
  ls -la /opt/tcp-analyzer/backend/
  ls -la /opt/tcp-analyzer/frontend/

  ---
  Phase 4 — Transfer the Gemma model

  4.1 — Create the model directory on the VM
  sudo mkdir -p /opt/models
  sudo chown $USER:$USER /opt/models

  4.2 — Transfer the model (run on your Windows machine in WSL or Git Bash — this will take a while)
  rsync -av --progress \
    "/mnt/c/Users/ppenthoi/.cache/huggingface/hub/models--google--gemma-4-12b-it/" \
    root@YOUR_VM_IP:/opt/models/gemma-4-12b-it/

  4.3 — Verify the model snapshot folder exists on the VM
  ls /opt/models/gemma-4-12b-it/snapshots/

  ---
  Phase 5 — Update the model path in the code

  5.1 — Check the current MODEL_ID line
  grep -n "MODEL_ID" /opt/tcp-analyzer/backend/app/routes/ai_chat.py

  5.2 — Replace the Windows path with the Linux path (copy the snapshot hash from Phase 4.3 output)
  sed -i 's|MODEL_ID = r"C:\\.*"|MODEL_ID = "/opt/models/gemma-4-12b-it/snapshots/5926caa4ec0cac5cbfadaf4077420520de1d5205"|' \
    /opt/tcp-analyzer/backend/app/routes/ai_chat.py

  5.3 — Confirm the change
  grep -n "MODEL_ID" /opt/tcp-analyzer/backend/app/routes/ai_chat.py

  ---
  Phase 6 — Set up Python virtual environment

  6.1 — Create the venv
  cd /opt/tcp-analyzer/backend
  python3.11 -m venv .venv

  6.2 — Activate it
  source /opt/tcp-analyzer/backend/.venv/bin/activate

  6.3 — Upgrade pip
  pip install --upgrade pip

  6.4 — Install PyTorch (CPU)
  pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu

  6.5 — Install the rest of the requirements
  pip install -r /opt/tcp-analyzer/backend/requirements.txt

  6.6 — Verify key packages loaded
  python -c "import fastapi, uvicorn, scapy, transformers, torch; print('All OK')"

  6.7 — Deactivate
  deactivate

  ---
  Phase 7 — Build the frontend

  7.1 — Install npm dependencies
  cd /opt/tcp-analyzer/frontend && npm install

  7.2 — Build production static files
  npm run build

  7.3 — Confirm dist folder was created
  ls -la /opt/tcp-analyzer/frontend/dist/

  ---
  Phase 8 — Create the systemd service

  8.1 — Create the service file (replace root with your actual Linux username if different)
  sudo tee /etc/systemd/system/tcp-analyzer-backend.service > /dev/null <<'EOF'
[Unit]
Description=TCP Analyzer Backend (FastAPI + Uvicorn)
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/tcp-analyzer/backend
ExecStart=/opt/tcp-analyzer/backend/.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000 --workers 1
Environment=CORS_ORIGINS=http://YOUR_VM_IP
Restart=always
RestartSec=5
StandardOutput=journal
StandardError=journal
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF

  8.2 — Reload systemd
  sudo systemctl daemon-reload

  8.3 — Enable the service
  sudo systemctl enable tcp-analyzer-backend

  8.4 — Start the service
  sudo systemctl start tcp-analyzer-backend

  8.5 — Check the status
  sudo systemctl status tcp-analyzer-backend

  8.6 — Check the logs
  sudo journalctl -u tcp-analyzer-backend -n 50 --no-pager

  ---
  Phase 9 — Configure nginx

  9.1 — Write the nginx config (replace YOUR_VM_IP)
  sudo tee /etc/nginx/conf.d/tcp-analyzer.conf > /dev/null <<'EOF'
  server {
      listen 80;
      server_name YOUR_VM_IP;

      root /opt/tcp-analyzer/frontend/dist;
      index index.html;

      location / {
          try_files $uri $uri/ /index.html;
      }

      location /api/ {
          proxy_pass http://127.0.0.1:8000;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
          client_max_body_size 500M;
          proxy_read_timeout    300s;
          proxy_connect_timeout  10s;
          proxy_send_timeout    300s;
      }
  }
  EOF

  9.2 — Test the nginx config
  sudo nginx -t

  ---
  Phase 10 — SELinux configuration

  10.1 — Allow nginx to proxy to the backend
  sudo setsebool -P httpd_can_network_connect 1

  10.2 — Label the frontend dist folder
  sudo chcon -Rt httpd_sys_content_t /opt/tcp-analyzer/frontend/dist/

  10.3 — Confirm the boolean was set
  getsebool httpd_can_network_connect

  ---
  Phase 11 — Start nginx

  11.1 — Enable nginx
  sudo systemctl enable nginx

  11.2 — Start nginx
  sudo systemctl restart nginx

  11.3 — Check nginx status
  sudo systemctl status nginx

  ---
  Phase 12 — Firewall

  12.1 — Enable firewalld
  sudo systemctl enable firewalld && sudo systemctl start firewalld

  12.2 — Open HTTP and SSH
  sudo firewall-cmd --permanent --add-service=http
  sudo firewall-cmd --permanent --add-service=ssh

  12.3 — Reload firewall
  sudo firewall-cmd --reload

  12.4 — Verify open services
  sudo firewall-cmd --list-all

  ---
  Phase 13 — Final verification

  13.1 — Health check from the VM itself
  curl -s http://localhost:8000/api/health

  13.2 — Frontend through nginx
  curl -I http://localhost

  13.3 — API through nginx
  curl -s http://localhost/api/health

  13.4 — From outside (run on your Windows machine)
  curl -s http://YOUR_VM_IP/api/health
